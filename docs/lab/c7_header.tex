\newcommand{\c@setpaper}[4]{% width, height, margin, extra height taken off the text block
  \pdfpagewidth=#1\relax \pdfpageheight=#2\relax
  \paperwidth=#1\relax \paperheight=#2\relax
  \textwidth=\dimexpr\paperwidth-2\dimexpr#3\relax\relax
  \textheight=\dimexpr\paperheight-2\dimexpr#3\relax-#4\relax
  \oddsidemargin=\dimexpr#3-1in\relax \evensidemargin=\oddsidemargin
  \topmargin=\dimexpr#3-1in-\headheight-\headsep\relax
  \hsize=\textwidth \columnwidth=\textwidth \linewidth=\textwidth
  \headwidth=\textwidth
  \global\@colht=\textheight \global\@colroom=\textheight \global\vsize=\textheight
}
